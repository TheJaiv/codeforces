def main():
    test_cases = int(input())
    for _ in range(test_cases):
        n = int(input())
        nums = list(map(int, input().split()))
        zeros = 0
        for i in nums:
            if i == 0:
                zeros += 1
 
        if zeros == 0:
            print('YES')
            print('A' * n)
        elif zeros == 1:
            print('NO')
        else:
            counter = 0
            ans = []
            for i in nums:
                if i == 0:
                    if counter == 0:
                        ans.append('A')
                        counter+=1
                    else:
                        ans.append('B')
                else:
                    ans.append('C')
            
            print("YES")
            print(''.join(ans))
 
 
 
main()
