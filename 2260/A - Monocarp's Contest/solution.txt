import sys
 
 
def input():
    return sys.stdin.readline().rstrip()
 
 
def main():
    test_cases = int(input())
    for _ in range(test_cases):
        n = int(input())
        tests = list(map(int, input().split(' ')))
        count = 0
        zeros = 0
        if n == 1:
            if tests[0] == 1:
                print(-1)
            else :
                print(0)
        elif n == 2:
            if 1 in tests:
                print(-1)
            else:
                print(0)
        else:
            if tests[0] == 1:
                count += 1
            if tests[n-1] == 1:
                count += 1
            for i in range(1,n-1):
                if tests[i] == 0:
                    zeros += 1
            if count > zeros:
                print(-1)
            else:
                print(count)
 
 
main()
