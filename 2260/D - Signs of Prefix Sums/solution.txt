import sys
 
 
def input():
    return sys.stdin.readline().rstrip()
 
 
def opposite(x, y):
    if x == '+' and y == '-':
        return True
    if x == '-' and y == '+':
        return True
    return False
 
 
def main():
    a = int(input())
    for _ in range(a):
        n = int(input())
        s = input()
 
        bad = False
        if s[0] == '0':
            bad = True
        for i in range(1, n):
            if s[i] == '0' and s[i - 1] == '0':
                bad = True
 
        if bad:
            print(-1)
            continue
 
        ans = 1
        i = 0
        while i < n:
            if s[i] == '0':
                i += 1
                continue
 
            j = i
            while j < n and s[j] == s[i]:
                j += 1
 
            length = j - i
 
            if i > 0:
                left = s[i - 1]
            else:
                left = '0'
 
            if j < n:
                right = s[j]
            else:
                right = 'E'
 
            if length == 2 and opposite(left, s[i]) and opposite(right, s[i]):
                ans = 3
            elif opposite(left, s[i]) or opposite(right, s[i]):
                if ans < 2:
                    ans = 2
            if right == '0' and length % 2 == 0:
                if ans < 2:
                    ans = 2
 
            i = j
 
        print(ans)
 
 
main()
