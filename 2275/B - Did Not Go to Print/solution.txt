import sys
 
 
def input():
    return sys.stdin.readline().rstrip()
 
 
def main():
    test_cases = int(input())
    for _ in range(test_cases):
        n = int(input())
        a = input()
        current = 1
        scanned = []
        printed = []
        for i in a:
            if int(i) == 1:
                scanned.append(current)
            if int(i) == 2:
                if len(scanned) != 0:
                    printed.append(scanned.pop())
                else:
                    printed.append(current)
            if int(i) == 3:
                printed.append(current)
            current +=1
        left = []
        scanned = []
        for i in range(0,n):
            scanned.append(i+1)
        left = list(set(scanned) - set(printed)  )
        print(len(left))
        print(' '.join(list(map(str, left))))
    
 
 
 
main()
