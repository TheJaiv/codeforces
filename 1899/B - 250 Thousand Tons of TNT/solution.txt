import sys
 
 
def input():
    return sys.stdin.readline().rstrip()
 
 
def main():
    test_cases = int(input())
    for _ in range(test_cases):
        n = int(input())
        boxes = list(map(int,input().split()))
        max_diff = 0
        for i in range(1,n+1):
            if i == 1:
                diff = max(boxes)-min(boxes)
                
                if diff > max_diff:
                    max_diff = diff
            elif n % i == 0:
                # print(str(i) + ".") 
                sum = []
                add = 0
                for j in range(0,n):
                    add += boxes[j]
                    sum.append(add)
                sums = []
                # print(sum) 
                sums.append(sum[i-1])
                #i=3 n=15
                for k in range(i-1,n-i ,i):
                    sums.append(sum[k+i]-sum[k])
                # print(sums)
                if len(sums) != 0:
                    diff = max(sums)-min(sums)
                    # print(str(diff) + "?")
                    if diff > max_diff:
                        max_diff = diff
        print(max_diff)
 
 
main()
