import sys
 
 
def input():
    return sys.stdin.readline().rstrip()
 
 
def main():
    test_cases = int(input())
    for _ in range(test_cases):
        a = input()
        b = input()
        max_same = 0
        for j in range(0,len(a)):
            for i in range(0,len(a)):
                # print(a[j:len(a)-i])
                if b.find(a[j:len(a)-i]) == -1:
                    continue
                else:
                    # print(a[j:len(a)-i])
                    same = len(a)-i-j
                    if same > max_same:
                        max_same = same
       
        # print(str(max_same) + ".")
        print(len(a)+len(b)-2*max_same)
 
main()
