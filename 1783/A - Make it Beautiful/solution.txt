import sys
 
 
def input():
    return sys.stdin.readline().rstrip()
 
 
def main():
    test_cases = int(input())
    for _ in range(test_cases):
        n = int(input())
        nums = list(map(int,input().split()))
        nums = sorted(nums)[::-1]
        if nums[0] == nums[1]:
            a = nums[0]
            nums[0] = nums[n-1]
            nums[n-1] = a
        if n != 1 and len(set(nums)) == 1:
            print("NO")
        else:
            print("YES")
            print(' '.join(list(map(str,nums))))
 
 
 
main()
