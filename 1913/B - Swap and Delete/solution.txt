test_cases = int(input())
for i in range(0,test_cases):
    s = input()
    zero = 0
    one = 0
    for i in s:
        if i == "0":
            zero += 1
        else :
            one += 1
    
    score = 0
    for i in s:
        if i == "0":
            if one > 0:
                one -= 1
            else:
                score = one + zero
                break
        else:
            if zero > 0:
                zero -= 1
            else :
                score = one + zero
                break
    print(score)
