import sys
 
 
def input():
    return sys.stdin.readline().rstrip()
 
 
def main():
    test_cases = int(input())
    for _ in range(test_cases):
        n = int(input())
        nums = list(map(int, input().split()))
        a = max(nums)
        remain = []
        ans = []
        dic = {}
        for i in nums:
            dic[i] = dic.get(i,0) + 1
        count = dic[a]
     
        dic1 = {}
        
 
        for i in sorted(nums)[::-1]:
            
            if (i == a):
                ans.append(i)
            else:
                if (dic1.get(i,0) < count):
                    ans.append(i)
                    dic1[i] = dic1.get(i,0) +1
                else:
                    remain.append(i)
        
        left = {}
        for i in remain:
            left[i] = left.get(i,0) + 1
        while left:
            for i in sorted(left)[::-1]:
                ans.append(i)
                left[i] -= 1
                if left[i] == 0:
                    del left[i]
 
        print(' '.join(list(map(str,ans))))
 
 
main()
