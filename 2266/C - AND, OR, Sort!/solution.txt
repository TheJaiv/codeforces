import sys
 
 
def input():
    return sys.stdin.readline().rstrip()
def check(n,s):
    
    dec = True
    for i in range(0,n-1):
        if int(s[i]) > int(s[i+1]):
            dec = False
            break
    return dec
 
 
def main():
    test_cases = int(input())
    for _ in range(test_cases):
        n = int(input())
        a = input()
        if check(n,a) == True:
            print(0)
            continue
        elif a[0] == '1':
            count = 0
            for i in a:
                if i == '0':
                    count += 1
            print(count)
            continue
 
        
        else:
            total_zeros = a.count('0')
            best = total_zeros
            ones = 0
            zeros_seen = 0
            for i in range(n):
                if a[i] == '1':
                    ones += 1
                else:
                    zeros_seen += 1
                best = min(best, ones + (total_zeros - zeros_seen))
            print(best)
 
 
 
 
 
main()
