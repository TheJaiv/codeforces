import sys
 
 
def input():
    return sys.stdin.readline().rstrip()
 
 
def main():
    test_cases = int(input())
    for _ in range(test_cases):
        n = int(input())
        nums = list(map(int,input().split()))
        left = 0
        right = n-1
        forward = nums[0]
        backward = nums[n-1]
        max = 0
        while (right > left):
            # print(left)
            # print(right)
            
            
            if forward > backward :
                right -= 1
                backward += nums[right]
            elif backward > forward:
                left += 1
                forward += nums[left]
            else:
                if (left+1 + (n-right)) > max:
                    max = left+1 + (n-right)
                    left += 1
                    right -= 1
                    backward += nums[right]
                    forward += nums[left]
            # print(".")
            # print(forward)
            # print(backward)
            # print(".")
        # print()
        print(max)
        # print()
 
 
 
 
main()
