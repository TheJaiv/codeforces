import sys
import math
 
def input():
    return sys.stdin.readline().rstrip()
 
def sumofsq(a):
    a = str(a)
    ans = 0
    for i in a:
        ans += (int(i))**2
    return ans
 
def main():
    test_cases = int(input())
    for _ in range(test_cases):
        n = int(input())
        nums = list(map(int,input().split()))
        a = [7,49,97,13,10,1]
        dic = {}
        for i in nums:
            m = i
            found = False
            for j in range(0,25):
                m = sumofsq(m)
                if m in a:
                    dic[1] = dic.get(1,0) + 1
                    found = True
                    break
            if found == False:
                dic[m] = dic.get(m,0) + 1
        pairs = 0
        for i in dic.values():
            if i > 1:
                pairs += math.comb(i,2)
        print(pairs)
 
main()
